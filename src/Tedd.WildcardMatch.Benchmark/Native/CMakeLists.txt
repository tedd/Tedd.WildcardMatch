cmake_minimum_required(VERSION 3.20)
project(wildcard_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(PCRE2_BUILD_PCRE2_8 ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_16 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_32 OFF CACHE BOOL "" FORCE)
set(PCRE2_SUPPORT_JIT ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2GREP OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/pcre2" pcre2)
add_library(wildcard_native SHARED native.cpp "${SOURCES}/git-standalone/wildmatch.c")
target_include_directories(wildcard_native PRIVATE "${SOURCES}/krauss" "${SOURCES}/wildcards/single_include" "${SOURCES}/git-standalone" "${CMAKE_CURRENT_BINARY_DIR}/pcre2")
target_link_libraries(wildcard_native PRIVATE pcre2-8-static)
if(MSVC)
  target_compile_options(wildcard_native PRIVATE /O2 /Ob2)
endif()
